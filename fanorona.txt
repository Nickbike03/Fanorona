% ==============================================================================
% GENERAZIONE MOSSE (Logica di base)
% ==============================================================================

potential_move_base(FX,FY, TX,TY, DX,DY) :-
    is_mine(FX,FY), valid_direction(FX,FY, DX,DY),
    TX = FX + DX, TY = FY + DY, is_empty(TX,TY).

% Calcolo Catture (Ricorsivo)
can_approach(FX,FY, TX,TY, DX,DY) :-
    potential_move_base(FX,FY, TX,TY, DX,DY),
    NX = TX + DX, NY = TY + DY, is_opp(NX,NY).

can_withdrawal(FX,FY, TX,TY, DX,DY) :-
    potential_move_base(FX,FY, TX,TY, DX,DY),
    NX = FX - DX, NY = FY - DY, is_opp(NX,NY).

% Conteggio Approach
seq_app(FX,FY, TX,TY, DX,DY, 1, NX,NY) :-
    can_approach(FX,FY, TX,TY, DX,DY), NX = TX + DX, NY = TY + DY.
seq_app(FX,FY, TX,TY, DX,DY, K1, NNX,NNY) :-
    seq_app(FX,FY, TX,TY, DX,DY, K, NX,NY),
    K1 = K + 1,
    NNX = NX + DX, NNY = NY + DY, is_opp(NNX,NNY).
max_app(FX,FY, TX,TY, N) :-
    seq_app(FX,FY, TX,TY, _, _, N, _, _),
    not exists_higher_app(FX,FY, TX,TY, N).
exists_higher_app(FX,FY, TX,TY, N) :-
    seq_app(FX,FY, TX,TY, _, _, N, _, _),
    seq_app(FX,FY, TX,TY, _, _, N2, _, _),
    N2 > N.

% Conteggio Withdrawal
seq_wit(FX,FY, TX,TY, DX,DY, 1, NX,NY) :-
    can_withdrawal(FX,FY, TX,TY, DX,DY), NX = FX - DX, NY = FY - DY.
seq_wit(FX,FY, TX,TY, DX,DY, K1, NNX,NNY) :-
    seq_wit(FX,FY, TX,TY, DX,DY, K, NX,NY),
    K1 = K + 1,
    NNX = NX - DX, NNY = NY - DY, is_opp(NNX,NNY).
max_wit(FX,FY, TX,TY, N) :-
    seq_wit(FX,FY, TX,TY, _, _, N, _, _),
    not exists_higher_wit(FX,FY, TX,TY, N).
exists_higher_wit(FX,FY, TX,TY, N) :-
    seq_wit(FX,FY, TX,TY, _, _, N, _, _),
    seq_wit(FX,FY, TX,TY, _, _, N2, _, _),
    N2 > N.

% Raggruppamento Mosse Valide con punteggio cattura
valid_move(FX,FY, TX,TY, approach, N) :- max_app(FX,FY, TX,TY, N).
valid_move(FX,FY, TX,TY, withdrawal, N) :- max_wit(FX,FY, TX,TY, N).
valid_move(FX,FY, TX,TY, paika, 0) :-
    potential_move_base(FX,FY, TX,TY, _,_),
    not any_capture_on_board.

% Flag: esiste almeno una cattura possibile sulla scacchiera?
any_capture_on_board :- valid_move(_,_,_,_,approach, _).
any_capture_on_board :- valid_move(_,_,_,_,withdrawal, _).
